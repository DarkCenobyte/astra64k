cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ASTRA_RELEASE LIMIT_COUNT 1)
project(Astra VERSION ${ASTRA_RELEASE} LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Astra's supported targets are 64-bit: x64 and ARM64")
endif()
configure_file(cmake/astra_version.h.in generated/astra_version.h @ONLY)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
option(ASTRA_BUILD_OFFLINE "Build the Linux OSMesa renderer" OFF)
option(ASTRA_BUILD_TESTS "Build shared-engine numerical/render tests" OFF)

if(WIN32)
  if(NOT MINGW OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "The 64k Windows build requires MinGW-w64 GCC 13+ (x86_64)")
  endif()
  if(CMAKE_C_COMPILER_VERSION VERSION_LESS 13)
    message(FATAL_ERROR "GCC 13+ is required for -Oz")
  endif()
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
    message(FATAL_ERROR "The Windows 64k target is x64")
  endif()
  add_executable(Astra WIN32 src/astra.c)
  set_target_properties(Astra PROPERTIES OUTPUT_NAME "Astra-v${PROJECT_VERSION}-windows-x64")
  target_compile_definitions(Astra PRIVATE __USE_MINGW_ANSI_STDIO=0)
  target_compile_options(Astra PRIVATE -Oz -ffast-math -fno-math-errno -fno-ident
    -fno-asynchronous-unwind-tables -fno-unwind-tables -fno-stack-protector)
  # Do not replace -nostartfiles with -nostdlib: v2-v5's missing math wrappers
  # caused direct CALLs into .idata. Reject implicit data auto-imports outright.
  target_link_options(Astra PRIVATE -s -nostartfiles
    "-Wl,--gc-sections,--entry=mainCRTStartup,--subsystem,windows,--no-insert-timestamp,--disable-auto-import,--fatal-warnings")
  target_link_libraries(Astra PRIVATE opengl32 gdi32 user32 winmm kernel32 msvcrt m)
  find_program(ASTRA_UPX NAMES upx upx-ucl REQUIRED)
  add_custom_command(TARGET Astra POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E env "ASTRA_OBJDUMP=${CMAKE_OBJDUMP}"
      "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check-pe.py" "$<TARGET_FILE:Astra>"
    COMMAND "${CMAKE_COMMAND}" -E copy "$<TARGET_FILE:Astra>" "$<TARGET_FILE_DIR:Astra>/Astra-unpacked.exe"
    COMMAND "${ASTRA_UPX}" --best --lzma "$<TARGET_FILE:Astra>"
    COMMAND "${ASTRA_UPX}" -t "$<TARGET_FILE:Astra>"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check-size.py" "$<TARGET_FILE:Astra>" 64000
    VERBATIM)
else()
  if(APPLE)
    # SDL is linked statically so the .app does not depend on a Homebrew prefix.
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
    set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
    set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
    set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL2
      URL https://github.com/libsdl-org/SDL/releases/download/release-2.32.10/SDL2-2.32.10.tar.gz
      URL_HASH SHA256=5f5993c530f084535c65a6879e9b26ad441169b3e25d789d83287040a9ca5165)
    FetchContent_MakeAvailable(SDL2)
    add_executable(Astra MACOSX_BUNDLE src/astra.c)
    set_target_properties(Astra PROPERTIES OUTPUT_NAME Astra
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
    target_compile_definitions(Astra PRIVATE GL_SILENCE_DEPRECATION)
    target_link_libraries(Astra PRIVATE SDL2-static "-framework OpenGL")
  else()
    find_package(SDL2 CONFIG REQUIRED)
    find_package(OpenGL REQUIRED)
    add_executable(Astra src/astra.c)
    target_link_libraries(Astra PRIVATE SDL2::SDL2 OpenGL::GL m)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
      set(ASTRA_ARCH arm64)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
      set(ASTRA_ARCH x64)
    else()
      message(FATAL_ERROR "Supported Linux targets: x64 and ARM64")
    endif()
    set_target_properties(Astra PROPERTIES OUTPUT_NAME "Astra-v${PROJECT_VERSION}-linux-${ASTRA_ARCH}")
  endif()
  target_compile_definitions(Astra PRIVATE ASTRA_SDL=1)
  target_compile_options(Astra PRIVATE -O2 -ffast-math)
endif()
target_include_directories(Astra PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

if(ASTRA_BUILD_OFFLINE OR ASTRA_BUILD_TESTS)
  if(WIN32 OR APPLE)
    message(FATAL_ERROR "OSMesa tools are an optional Linux-only build")
  endif()
  find_path(ASTRA_OSMESA_INCLUDE GL/osmesa.h REQUIRED)
  find_library(ASTRA_OSMESA_LIBRARY OSMesa REQUIRED)
  if(ASTRA_BUILD_OFFLINE)
    add_executable(astra-render src/astra.c)
    add_executable(astra-stills tools/render-stills.c)
    foreach(target astra-render astra-stills)
      target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${ASTRA_OSMESA_INCLUDE}" "${OPENGL_INCLUDE_DIR}")
      target_compile_options(${target} PRIVATE -O2 -ffast-math)
      target_link_libraries(${target} PRIVATE "${ASTRA_OSMESA_LIBRARY}" m)
    endforeach()
  endif()
  if(ASTRA_BUILD_TESTS)
    enable_testing()
    add_executable(astra-engine-check tools/check-scenes.c)
    target_include_directories(astra-engine-check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${ASTRA_OSMESA_INCLUDE}" "${OPENGL_INCLUDE_DIR}")
    # Keep asserts and finite-value checks even with CMAKE_BUILD_TYPE=Release.
    target_compile_options(astra-engine-check PRIVATE -O2 -UNDEBUG)
    target_link_libraries(astra-engine-check PRIVATE "${ASTRA_OSMESA_LIBRARY}" m)
    add_test(NAME engine COMMAND astra-engine-check)
    set_tests_properties(engine PROPERTIES TIMEOUT 180)
  endif()
endif()
